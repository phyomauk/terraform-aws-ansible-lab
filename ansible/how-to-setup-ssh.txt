# check if the agent is running
eval "$(ssh-agent -s)"

### how to add the private key to the ssh agent
# For default RSA key
ssh-add ~/.ssh/id_rsa

# For ED25519 key
ssh-add ~/.ssh/id_ed25519

# For custom named AWS key (e.g., my-aws-key.pem)
ssh-add ~/.ssh/my-aws-key.pem

### how to verify the key
ssh-add -l
